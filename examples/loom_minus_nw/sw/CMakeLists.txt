cmake_minimum_required(VERSION 3.5)
project(example_loom_minus_nw_sw)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(EN_SIM "Enable simulation mode for Coyote" OFF)
if(EN_SIM)
    add_subdirectory(../../../sim/sw ${CMAKE_BINARY_DIR}/coyote)
else()
    add_subdirectory(../../../sw ${CMAKE_BINARY_DIR}/coyote)
endif()
# loom.hpp is shared with examples/loom rather than copied
include_directories(${CMAKE_SOURCE_DIR}/../../loom/sw/src)
add_executable(mnw_bench ${CMAKE_SOURCE_DIR}/src/mnw_bench.cpp)
target_link_libraries(mnw_bench PUBLIC Coyote)
